static void analysisScript(void)
{
	BOOL result, label_jump;
	u8 label_num, i, n, next_check;

	result = TRUE;
	label_jump = FALSE;
	next_check = 0;
	if (g_script_start)
	{
		label_num = g_start_label;
		label_jump = TRUE;
		g_script_start = FALSE;
	}
	setChangeLock(TRUE);
	while (result)
	{
		switch (*g_script_ptr)
		{
			case SCRIPT_COMMAND_LABEL:			// ラベル(ローカルのみ) byte:ラベル番号
				if (label_jump)
				{
					if (*(g_script_ptr+1) == label_num)
						label_jump = FALSE;
				}
				g_script_ptr += 2;		// スルー
				break;
0			case SCRIPT_COMMAND_GOTO:			// ラベルジャンプ byte:ラベル番号
				if (!label_jump)
				{
					label_num = *(g_script_ptr+1);
					label_jump = TRUE;
					g_script_ptr = (u8 *)g_script_ptr_base;
				} else
					g_script_ptr += 2;		// スルー
				break;
1			case SCRIPT_COMMAND_FLAG:			// フラグセット word:フラグ番号 byte:ON/OFF
				if (!label_jump)
				{
					u16 num;
					BOOL flag;
					memcpyBDS((s8 *)&num, (s8 *)g_script_ptr+1, 2);
					if (*(g_script_ptr+3) != 0)
						flag = TRUE;
					else
						flag = FALSE;
					setLabel(num, flag);
				}
				g_script_ptr += 4;
				break;
2			case SCRIPT_COMMAND_BG:				// BG変更 word:BG番号
				if (!label_jump)
				{
					u16 num;

					if (next_check != 0 && next_check != 4)
					{
						result = FALSE;
						break;
					}
					memcpyBDS((s8 *)&num, (s8 *)g_script_ptr+1, 2);
					setBgNum(num);
					next_check = 1;
				}
				g_script_ptr += 3;
				break;
3			case SCRIPT_COMMAND_BG2:			// BG変更 word:BG番号
				if (!label_jump)
				{
					u16 num;

					if (next_check != 0)
					{
						result = FALSE;
						break;
					}
					memcpyBDS((s8 *)&num, (s8 *)g_script_ptr+1, 2);
					setBgNum2(num);
					result = FALSE;
				}
				g_script_ptr += 3;
				break;
4			case SCRIPT_COMMAND_CHAR:			// CHA変更 byte:位置 word:CHA番号
				if (!label_jump)
				{
					u16 num;

					if (next_check != 0 && next_check != 1)
					{
						result = FALSE;
						break;
					}
					memcpyBDS((s8 *)&num, (s8 *)g_script_ptr+2, 2);
					setChaNum(*(g_script_ptr+1), num);
					next_check = 4;
				}
				g_script_ptr += 4;
				break;
5			case SCRIPT_COMMAND_OFF_CHAR:		// CHA消去
				if (!label_jump)
				{
					if (next_check != 0 && next_check != 2)
					{
						result = FALSE;
						break;
					}
					clearCha(*(g_script_ptr+1));
					next_check = 2;
				}
				g_script_ptr += 2;
				break;
6			case SCRIPT_COMMAND_FACE:			// FACE変更 word:FACE番号
				if (!label_jump)
				{
					u8 nc;
					u16 num;

					memcpyBDS((s8 *)&num, (s8 *)g_script_ptr+1, 2);
					if (num != 0xffff)
						nc = 1;
					else
						nc = 2;
					if (next_check != 0 && next_check != nc && !(nc == 1 && next_check == 4))
					{
						result = FALSE;
						break;
					}
					setFace(num);
					next_check = nc;
				}
				g_script_ptr += 3;
				break;
7			case SCRIPT_COMMAND_VOICE:			// ボイス再生 dword:ボイス番号
				if (!label_jump)
				{
					u16 num;

					memcpyBDS((s8 *)&num, (s8 *)g_script_ptr+1, 2);
					playVoice(num);
					g_voice_play = TRUE;
				}
				g_script_ptr += 3;
				break;
8			case SCRIPT_COMMAND_TEXT_MODE:		// テキストの表示スイッチ byte:0/ウィンドウ 1/全画面
				if (!label_jump)
				{
					g_text_mode = *(g_script_ptr+1);
					// テキストの初期位置
					g_text_x = SCRIPT_TEXT_START_X;
					g_text_y = SCRIPT_TEXT_START_Y;
				}
				g_script_ptr += 2;
				break;
9			case SCRIPT_COMMAND_MESSAGE_NAME:	// FACE欄の名前の設定 byte:文字列長 str:名前
				n = *(g_script_ptr+1);
				if (!label_jump)
				{
					char code[6*2];
					u8 len;

					if (next_check != 0)
					{
						result = FALSE;
						break;
					}
					memcpyBDS((s8 *)code, (s8 *)(g_script_ptr+2), (u32)(*(g_script_ptr+1)*2));
					len = changeCode((u16 *)code, *(g_script_ptr+1));
					setFaceNameCode(len*2, (char *)code);
					result = FALSE;
				}
				g_script_ptr += (n*2)+2;
				break;
a			case SCRIPT_COMMAND_MESSAGE:		// メッセージ表示 byte:文字列長 str:文字列
				if (!label_jump)
				{
					char code[20*2];
					u8 len;

					if (next_check != 0 && next_check != 3)
					{
						result = FALSE;
						break;
					}
					memcpyBDS((s8 *)code, (s8 *)(g_script_ptr+2), (u32)(*(g_script_ptr+1)*2));
					len = changeCode((u16 *)code, *(g_script_ptr+1));
					if (g_text_mode == 0)
					{
						// ウィンドウ表示
						setMessageCode(g_message_line, len*2, (char *)code);
						g_step = SCRIPT_STEP_MES_WAIT;
					} else {
						// 全画面表示
						setCodeToBmpPtrPos(getBmpBuf(), (u8 *)code, len*2, 0, SCRIPT_TEXT_START_X, g_text_y);
						g_text_x = (s16)(SCRIPT_TEXT_START_X + len * 12);
					}
					next_check = 3;
				}
				n = *(g_script_ptr+1);
				g_script_ptr += (n*2) + 2;
				break;
b			case SCRIPT_COMMAND_LINE_FEED:		// 改行
				if (!label_jump)
				{
					if (next_check != 0 && next_check != 3)
					{
						result = FALSE;
						break;
					}
					if (g_text_mode == 0)
					{
						// ウィンドウ表示
						g_message_line++;
						if (g_message_line > 2)
							g_message_line = 0;
					} else {
						// 全画面表示
						g_text_x = SCRIPT_TEXT_START_X;
						g_text_y += 14;
					}
					next_check = 3;
				}
				g_script_ptr += 1;
				break;
c			case SCRIPT_COMMAND_KEY_WAIT:		// キーウェイト
				if (!label_jump)
				{
					if (next_check != 0)
					{
						result = FALSE;
						break;
					}
					if (g_text_mode == 0)
						setKeyWaitPos(-1, -1);
					else
						setKeyWaitPos(g_text_x-4, g_text_y-3);
					//if (!(getPad() & PAD_BUTTON_X))
					//{
						putKeyWait();
						g_step = SCRIPT_STEP_KEY_WAIT;
					//} else {
					//	clearMessage();
					//	g_message_line = 0;
					//}
					result = FALSE;
				}
				g_script_ptr += 1;
				break;
d			case SCRIPT_COMMAND_MESSAGE_WINDOW:	// メッセージウィンドウON/OFF byte:ON/OFF
				if (!label_jump)
				{
					u8 nc;
					BOOL put;
					if (*(g_script_ptr+1) == 1)
					{
						nc = 1;
						put = TRUE;
					} else {
						nc = 2;
						put = FALSE;
					}
					if (next_check != 0 && next_check != nc)
					{
						result = FALSE;
						break;
					}
					if (g_text_mode == 0)
						putMessageWindow(put);
					else
					if (!put)
					{
						updateGraph();
						g_text_x = SCRIPT_TEXT_START_X;
						g_text_y = SCRIPT_TEXT_START_Y;
					}
					next_check = nc;
				}
				g_script_ptr += 2;
				break;
e			case SCRIPT_COMMAND_DATE_WINDOW:	// 日付ウィンドウON/OFF byte:ON/OFF
				if (!label_jump)
				{
				}
				g_script_ptr += 2;
				break;
f			case SCRIPT_COMMAND_OFF_FACE_WINDOW:	// FACEウィンドウOFF
				if (!label_jump)
				{
					if (next_check != 0 && next_check != 2)
					{
						result = FALSE;
						break;
					}
					clearFaceName();
					putFaceWindow(FALSE);
					next_check = 2;
				}
				g_script_ptr += 1;
				break;
10			case SCRIPT_COMMAND_SELECT:			// 選択肢の表示
				if (!label_jump)
				{
					char code[12*2];
					u8 len;

					if (next_check != 0)
					{
						result = FALSE;
						break;
					}
					initSelect();
					n = 2;
					for (i=0; i<*(g_script_ptr+1); i++)
					{
						g_select_label[i] = *(g_script_ptr+n);
						memcpyBDS((s8 *)code, (s8 *)(g_script_ptr+n+2), (u32)(*(g_script_ptr+n+1)*2));
						len = changeCode((u16 *)code, *(g_script_ptr+n+1));
						setSelectCode(i, len*2, (char *)code);
						n += (u8)((*(g_script_ptr+n+1) * 2) + 2);
					}
					putStartSelect();
					g_step = SCRIPT_STEP_SEL_WAIT;
					result = FALSE;
				}
				n = 2;
				for (i=0; i<*(g_script_ptr+1); i++)
				{
					n++;
					n += (u8)((*(g_script_ptr+n) * 2) + 1);
				}
				g_script_ptr += n;
				break;

11			case SCRIPT_COMMAND_SE_PLAY:			// 
				if (!label_jump)
				{
					if (getOptionSFX() != 0)
						playAdxM(g_se_table[*(g_script_ptr+1)]);
				}
				g_script_ptr += 2;
				break;

12			case SCRIPT_COMMAND_SE_WAIT:			// 
				if (!label_jump)
				{
					if (next_check != 0)
					{
						result = FALSE;
						break;
					}
					g_step = SCRIPT_STEP_SE_WAIT;
					result = FALSE;
				}
				g_script_ptr += 1;
				break;

13			case SCRIPT_COMMAND_SE_STOP:			// 
				if (!label_jump)
				{
					stopAdxM();
				}
				g_script_ptr += 1;
				break;

14			case SCRIPT_COMMAND_BGM_PLAY:		// 
				if (!label_jump)
				{
					int n;
					n = g_bgm_table[*(g_script_ptr+1)];
					setBGM(n);
					playBGM();
				}
				g_script_ptr += 2;
				break;

15			case SCRIPT_COMMAND_ROOM_BGM_PLAY:		// 
				if (!label_jump)
					playRoomBGM();
				g_script_ptr += 1;
				break;

16			case SCRIPT_COMMAND_BGM_STOP:		// 
				if (!label_jump)
				{
					u16 num;

					if (next_check != 0)
					{
						result = FALSE;
						break;
					}
					memcpyBDS((s8 *)&num, (s8 *)g_script_ptr+1, 2);
					fadeBGM(num);
					if (num > 0)
					{
						g_fade_time = num;
						g_step = SCRIPT_STEP_BGM_WAIT;
						result = FALSE;
					}
				}
				g_script_ptr += 3;
				break;

17			case SCRIPT_COMMAND_SPECIAL:		// 
				if (!label_jump)
				{
					if (next_check != 0)
					{
						result = FALSE;
						break;
					}
					g_special = *(g_script_ptr+1);
					g_step = SCRIPT_STEP_SPECIAL;
					result = FALSE;
				}
				g_script_ptr += 2;
				break;

18			case SCRIPT_COMMAND_QUAKE:			// 
				if (!label_jump)
				{
					if (next_check != 0)
					{
						result = FALSE;
						break;
					}
					setQuake();
					g_step = SCRIPT_STEP_QUAKE_WAIT;
					result = FALSE;
				}
				g_script_ptr += 1;
				break;

19			case SCRIPT_COMMAND_GOLD:			// 
				if (!label_jump)
				{
				}
				break;
1a			case SCRIPT_COMMAND_PARAM:			// 
				if (!label_jump)
				{
				}
				break;

1b			case SCRIPT_COMMAND_END:
				if (!label_jump)
				{
					//clearMessage();
					g_script_end_code = *(g_script_ptr+1);
					if (g_script_ptr_base)
					{
						freeBDS(g_script_ptr_base);
						g_script_ptr_base = NULL;
					}
					g_script_exec = FALSE;
					g_step = SCRIPT_STEP_NONE;
					result = FALSE;
				} else
					g_script_ptr += 2;
				break;
			default:
				;
		}
	}
	if (next_check == 3)
	{
		if (g_step == SCRIPT_STEP_MES_WAIT)
			putStartMessage();
		if (g_text_mode != 0)
		{
			if (getPad() & PAD_BUTTON_X)
				flipBmpBuf(0);
			else
			if (getOptionMessage() == 0)
				flipBmpBuf(10);
			else
				flipBmpBuf(30);
		}
	}
	setChangeLock(FALSE);
}
